#include <WiFi.h>             // thư viện dùng để kết nối Wifi của ESP32
#include <WiFiClientSecure.h> // thư viện dùng để tạo kết nối http/mqtt qua TLS
#include <PubSubClient.h>     // thư viện dùng để giao tiếp với mqtt broker
#include <ArduinoJson.h>      // thư viện dùng để tạo và xử lý dữ liệu JSON
#include <OneWire.h>          // thư viện dùng cho cảm biến nhiệt độ
#include <DallasTemperature.h>// thư viện dùng cho cảm biến nhiệt độ

// ===== WiFi Info =====
// cấu hình wifi defaul (chưa làm tính năng cho người dùng tự cấu hình)
const char* ssid = "CLCK23";      
const char* password = "02092023";

// ===== MQTT Info =====
const char* mqtt_server = "195d12f952ea4bbba0db56a9e044028f.s1.eu.hivemq.cloud";
const int mqtt_port = 8883;  // TLS port
const char* mqtt_user = "NPT100";
const char* mqtt_pass = "Phuctho100";
const char* mqtt_topic = "sensor/data";

// ===== Sensors Setup =====
#define MQ2_PIN     32
#define MQ7_PIN     33
#define MQ135_PIN   35
#define FLAME_PIN   34
#define ONE_WIRE_BUS 4


// Khởi tạo giao tiếp với cảm biển nhiệt độ
OneWire oneWire(ONE_WIRE_BUS);
DallasTemperature sensors(&oneWire);

// ===== MQTT & WiFiClientSecure =====
WiFiClientSecure wifiClient;    // tạo kết nối bảo mật TLS
PubSubClient client(wifiClient);// MQTT client sử dụng kết nói bảo mật, sử dụng wifiClient làm kênh kết nối

// ===== Certificate check (optional) =====
void setup_wifi() {
  Serial.print("🔌 Đang kết nối WiFi");
  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }
  Serial.println("\n✅ Đã kết nối WiFi");

  wifiClient.setInsecure(); // ⚠️ Không kiểm tra chứng chỉ (nếu cần bảo mật cao, bạn cần tải chứng chỉ và dùng `wifiClient.setCACert(...)`)
}

void reconnect() {
  while (!client.connected()) {
    Serial.print("🔄 Kết nối MQTT TLS...");
    String clientId = "ESP32Client-" + String(random(0xffff), HEX);
    if (client.connect(clientId.c_str(), mqtt_user, mqtt_pass)) {
      Serial.println("✅ MQTT đã kết nối");
    } else {
      Serial.print("❌ Lỗi MQTT: ");
      Serial.print(client.state());
      Serial.println(" → thử lại sau 5s");
      delay(5000);
    }
  }
}

void setup() {
  Serial.begin(115200);
  sensors.begin();
  pinMode(FLAME_PIN, INPUT);
  
  // chân analog sẽ tự nhận biết INPUT hay OUTPUT thông qua lệnh read hoặc write nên không cần khai báo

  setup_wifi();                             // kết nối wifi
  client.setServer(mqtt_server, mqtt_port); // cấu hình địa chỉ MQTT
}

void loop() {
  if (!client.connected()) {
    reconnect(); // tự động kết nối lại với mqtt nếu mất kết nối
  }
  client.loop(); // duy trì kết nối

  // Đọc cảm biến
  sensors.requestTemperatures();
  float temperature = sensors.getTempCByIndex(0);
  int mq2 = analogRead(MQ2_PIN);
  int mq7 = analogRead(MQ7_PIN);
  int mq135 = analogRead(MQ135_PIN);
  int flame = digitalRead(FLAME_PIN);

  // In ra Serial
  Serial.printf("🌡 Temp: %.2f | MQ2: %d | MQ7: %d | MQ135: %d | Flame: %s\n",
                temperature, mq2, mq7, mq135, flame == LOW ? "🔥" : "OK");

  // Tạo JSON payload
  StaticJsonDocument<256> doc;
  doc["temperature"] = temperature;
  doc["mq2"] = mq2;
  doc["mq7"] = mq7;
  doc["mq135"] = mq135;
  doc["flame"] = flame;

  char payload[256];
  serializeJson(doc, payload);

  // Gửi lên topic MQTT
  client.publish(mqtt_topic, payload);

  delay(10000); // mỗi 10 giây
}
