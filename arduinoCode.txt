#include <WiFi.h>
#include <HTTPClient.h>
#include <ArduinoJson.h>
#include <OneWire.h>
#include <DallasTemperature.h>

// ==== WiFi & Backend Setup ====
const char* ssid = "CLCK23";
const char* password = "02092023";
const char* backendIP = "192.168.1.132"; // Đổi thành IP backend thực tế

String jwtToken = "";

// ==== Sensor Pins ====
#define MQ2_PIN     32
#define MQ7_PIN     33
#define MQ135_PIN   35
#define FLAME_PIN   34
#define ONE_WIRE_BUS 4

OneWire oneWire(ONE_WIRE_BUS);
DallasTemperature sensors(&oneWire);

// ==== Hàm login để lấy token ====
bool loginAndGetToken() {
  if (WiFi.status() != WL_CONNECTED) return false;

  HTTPClient http;
  http.begin("http://" + String(backendIP) + ":8080/api/auth/login");
  http.addHeader("Content-Type", "application/json");

  // ✅ Đảm bảo dùng đúng thông tin tài khoản (username hoặc email đều được nếu backend hỗ trợ)
  String loginPayload = "{\"username\":\"nptho23@clc.fitus.edu.vn\",\"password\":\"123\"}";

  int responseCode = http.POST(loginPayload);
  if (responseCode == 200 || responseCode == 201) {
    String res = http.getString();
    Serial.println("✅ Login response: " + res);

    DynamicJsonDocument doc(512);
    DeserializationError err = deserializeJson(doc, res);
    if (err) {
      Serial.println("❌ JSON parse error: " + String(err.c_str()));
      http.end();
      return false;
    }

    jwtToken = doc["access_token"].as<String>();
    Serial.println("✅ Token saved");
    http.end();
    return true;
  } else {
    Serial.println("❌ Login failed. Code: " + String(responseCode));
    http.end();
    return false;
  }
}

// ==== Gửi dữ liệu cảm biến ====
void sendSensorData() {
  if (WiFi.status() != WL_CONNECTED) return;

  if (jwtToken == "") {
    Serial.println("🔄 Token rỗng, đang đăng nhập...");
    if (!loginAndGetToken()) {
      Serial.println("❌ Không thể lấy token");
      return;
    }
  }

  // Đọc dữ liệu cảm biến
  sensors.requestTemperatures();
  float temperature = sensors.getTempCByIndex(0);
  int mq2 = analogRead(MQ2_PIN);
  int mq7 = analogRead(MQ7_PIN);
  int mq135 = analogRead(MQ135_PIN);
  int flame = digitalRead(FLAME_PIN);

  Serial.println("---- 🔍 Dữ liệu cảm biến ----");
  Serial.printf("🌡 Nhiệt độ: %.2f °C | MQ2: %d | MQ7: %d | MQ135: %d | Lửa: %s\n",
                temperature, mq2, mq7, mq135, flame == LOW ? "🔥 Có lửa" : "An toàn");

  // Gửi lên server
  HTTPClient http;
  http.begin("http://" + String(backendIP) + ":8080/api/in4-arduino/upload");
  http.addHeader("Content-Type", "application/json");
  http.addHeader("Authorization", "Bearer " + jwtToken);

  DynamicJsonDocument payload(512);
  payload["mq2"] = mq2;
  payload["mq7"] = mq7;
  payload["mq135"] = mq135;
  payload["temperature"] = temperature;
  payload["flame"] = flame;

  String body;
  serializeJson(payload, body);
  int code = http.POST(body);

  if (code == 401) {
    Serial.println("🔒 Token hết hạn hoặc sai. Đang đăng nhập lại...");
    jwtToken = "";
    loginAndGetToken(); // thử đăng nhập lại
  } else {
    Serial.println("📤 Mã phản hồi gửi dữ liệu: " + String(code));
  }

  http.end();
}

// ==== Setup ====
void setup() {
  Serial.begin(115200);
  sensors.begin();
  pinMode(FLAME_PIN, INPUT);

  Serial.print("🔌 Đang kết nối WiFi");
  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) {
    Serial.print(".");
    delay(500);
  }
  Serial.println("\n✅ Đã kết nối WiFi");

  loginAndGetToken();
}

// ==== Loop ====
void loop() {
  sendSensorData();
  delay(10000); // mỗi 10 giây
}
